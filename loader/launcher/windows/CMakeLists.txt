cmake_minimum_required(VERSION 3.21)

add_library(ProxyLoader SHARED proxyLoader.cpp)
target_compile_features(ProxyLoader PUBLIC cxx_std_23)
set_target_properties(ProxyLoader PROPERTIES
	PREFIX ""
	OUTPUT_NAME "XInput1_4"
	LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NOAHH_BIN_PATH}/nightly"
	RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NOAHH_BIN_PATH}/nightly"
	ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NOAHH_BIN_PATH}/nightly"
	LIBRARY_OUTPUT_DIRECTORY_RELEASE "${NOAHH_BIN_PATH}/nightly"
	RUNTIME_OUTPUT_DIRECTORY_RELEASE "${NOAHH_BIN_PATH}/nightly"
	ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${NOAHH_BIN_PATH}/nightly"
	LIBRARY_OUTPUT_DIRECTORY_MINSIZEREL "${NOAHH_BIN_PATH}/nightly"
	RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL "${NOAHH_BIN_PATH}/nightly"
	ARCHIVE_OUTPUT_DIRECTORY_MINSIZEREL "${NOAHH_BIN_PATH}/nightly"
	LIBRARY_OUTPUT_DIRECTORY "${NOAHH_BIN_PATH}/nightly"
	RUNTIME_OUTPUT_DIRECTORY "${NOAHH_BIN_PATH}/nightly"
	ARCHIVE_OUTPUT_DIRECTORY "${NOAHH_BIN_PATH}/nightly"
)

string(REPLACE "." "_" NOAHH_GD_VERSION_IDENTIFIER "_${NOAHH_GD_VERSION}")
target_compile_definitions(ProxyLoader PRIVATE NOAHH_GD_VERSION_IDENTIFIER=${NOAHH_GD_VERSION_IDENTIFIER})

add_executable(Updater Updater.cpp)
target_compile_features(Updater PUBLIC cxx_std_23)
set_target_properties(Updater PROPERTIES
	PREFIX ""
	OUTPUT_NAME "NoahhUpdater"
	RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NOAHH_BIN_PATH}/nightly"
	ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NOAHH_BIN_PATH}/nightly"
	LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NOAHH_BIN_PATH}/nightly"
	LIBRARY_OUTPUT_DIRECTORY_RELEASE "${NOAHH_BIN_PATH}/nightly"
	RUNTIME_OUTPUT_DIRECTORY_RELEASE "${NOAHH_BIN_PATH}/nightly"
	ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${NOAHH_BIN_PATH}/nightly"
	LIBRARY_OUTPUT_DIRECTORY_MINSIZEREL "${NOAHH_BIN_PATH}/nightly"
	RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL "${NOAHH_BIN_PATH}/nightly"
	ARCHIVE_OUTPUT_DIRECTORY_MINSIZEREL "${NOAHH_BIN_PATH}/nightly"
	LIBRARY_OUTPUT_DIRECTORY "${NOAHH_BIN_PATH}/nightly"
	RUNTIME_OUTPUT_DIRECTORY "${NOAHH_BIN_PATH}/nightly"
	ARCHIVE_OUTPUT_DIRECTORY "${NOAHH_BIN_PATH}/nightly"
)

if (MSVC)
	include(CheckLinkerFlag)
	check_linker_flag(CXX /NOEXP SUPPORTS_NOEXP)
	if (SUPPORTS_NOEXP)
		target_link_options(ProxyLoader PRIVATE /NOEXP)
	endif()

	target_link_options(ProxyLoader PRIVATE /NOIMPLIB /DEBUG:NONE)
	target_link_options(Updater PRIVATE /DEBUG:NONE)
endif()
